    <script>
        // 4. TIMELINE
        const tl = gsap.timeline({ defaults: { duration: 0.8, ease: "power2.out" } });

        // 4.1.POSITION PARAMETER
        tl.from("h1", { y: -40, opacity: 0 })
            .from("nav a", { y: -20, opacity: 0, stagger: 0.15 }, "-=0.4")

        // 4.2. CONTROLLING TIMELINE
        const replayBtn = document.querySelector("#replayBtn");

        replayBtn.addEventListener("click", function () {
            tl.restart();
        });

        gsap.registerPlugin(ScrollTrigger);

        const reveals = gsap.utils.toArray(".reveal");

        reveals.forEach(function (el) {
            gsap.from(el, {
                y: 50,
                opacity: 0,
                duration: 0.8,
                ease: "power2.out",
                scrollTrigger: {
                    trigger: el,
                    start: "top 80%",
                    markers: true,
                    toggleActions: "play none none reverse"
                }
            });
        });

        gsap.fromTo("#progressBar",
            { scaleX: 0 },
            {
                scaleX: 1,
                ease: "none",
                scrollTrigger: {
                    trigger: "body",
                    start: "top top",
                    end: "bottom bottom",
                    scrub: true
                }
            }
        );

        ScrollTrigger.create({
            trigger: "#services",
            start: "top top",
            end: "+=400",
            pin: true,
            markers: true
        });
    </script>
    <script>
        const darkModeToggle = document.querySelector("#darkModeToggle");
        const html = document.querySelector("html");
        let spinTween;

        darkModeToggle.addEventListener("click", function () {
            if (!spinTween) {
                // Start slowly, accelerate, then keep spinning until the page reloads.
                spinTween = gsap.to(darkModeToggle, {
                    rotation: "+=360",
                    duration: 2,
                    ease: "power1.in",
                    repeat: -1,
                    onRepeat() {
                        this.duration(Math.max(0.08, this.duration() * 0.8));
                    }
                });
            }
            html.classList.toggle("dark");
        });
    </script>
